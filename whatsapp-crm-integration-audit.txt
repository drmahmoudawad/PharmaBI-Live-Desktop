async function autoConnectSavedBranches() {

    let connections =
        [];


    try {

        connections =
            getAllSavedConnections();

    }
    catch (
        error
    ) {

        console.error(

            "[Electron] Failed to load saved connections:",

            error

        );


        return {

            success:
                false,

            connected:
                0,

            failed:
                0

        };

    }


    if (
        !Array.isArray(
            connections
        ) ||
        connections.length ===
        0
    ) {

        console.log(

            "[Electron] No saved branch connections."

        );


        return {

            success:
                true,

            connected:
                0,

            failed:
                0

        };

    }


    console.log(

        `[Electron] Auto-connecting ${connections.length} saved branch(es)...`

    );


    let connected =
        0;


    let failed =
        0;


    for (
        const connection
        of connections
    ) {

        const branchId =
            Number(
                connection?.branchId
            );


        if (
            !Number.isInteger(
                branchId
            ) ||
            branchId <= 0
        ) {

            failed +=
                1;


            continue;

        }


        try {

            const connectPromise =
                getBranchPool(
                    branchId,
                    connection
                );

            const connectTimeout =
                new Promise(
                    (_, reject) => {
                        setTimeout(
                            () =>
                                reject(
                                    new Error(
                                        `Auto-connect timeout after 300000ms for branch ${branchId}`
                                    )
                                ),
                            300000
                        );
                    }
                );

            await Promise.race([
                connectPromise,
                connectTimeout
            ]);


            connected +=
                1;


            console.log(

                `[Electron] Branch ${branchId} connected automatically.`

            );

        }
        catch (
            error
        ) {

            failed +=
                1;


            console.error(

                `[Electron] Branch ${branchId} auto-connect failed:`,

                error

            );

        }

    }


    console.log(

        `[Electron] Auto-connect complete. connected=${connected} failed=${failed}`

    );


    return {

        success:
            failed === 0,

        connected,

        failed

    };

}

async function startBackgroundWarmup() {

    console.log(
        "[Electron] Initial current-month warmup delayed 300000ms to prioritize UI loading."
    );

    await new Promise(
        resolve =>
            setTimeout(
                resolve,
                300000
            )
    );

    try {

        const result =
            await refreshCurrentMonthAllBranches({

                maxRetries:
                    2

            });

        console.log(
            "[Electron] Initial current-month refresh result:",
            result
        );

    }
    catch (
        error
    ) {

        console.error(
            "[Electron] Initial current-month refresh failed:",
            error
        );

    }

    if (
        !currentMonthRefreshTimer
    ) {

        currentMonthRefreshTimer =
            setInterval(
                async () => {

                    try {

                        const result =
                            await refreshCurrentMonthAllBranches({

                                maxRetries:
                                    2

                            });

                        console.log(
                            "[Electron] 10-minute current-month refresh result:",
                            result
                        );

                    }
                    catch (
                        error
                    ) {

                        console.error(
                            "[Electron] 10-minute current-month refresh failed:",
                            error
                        );

                    }

                },
                10 *
                60 *
                1000
            );

        console.log(
            "[Electron] Current-month auto-refresh scheduled every 10 minutes."
        );

    }

}

let crmPhoneSyncTimer =
    null;

let crmPhoneSyncRunning =
    false;


async function runCrmPhoneAutoSync() {

    if (
        crmPhoneSyncRunning
    ) {
        return;
    }

    crmPhoneSyncRunning =
        true;

    try {

        const branches =
            await getAllSavedConnections();

        if (
            !Array.isArray(
                branches
            )
        ) {
            return;
        }

        const results =
            await Promise.allSettled(
                branches.map(
                    async branch => {

                        const branchId =
                            Number(
                                branch?.branchId
                            );

                        if (
                            !Number.isInteger(
                                branchId
                            ) ||
                            branchId <= 0
                        ) {
                            return null;
                        }

                        try {

                            return await syncCrmPhonesForBranch(
                                branchId
                            );

                        } catch (
                            error
                        ) {

                            console.warn(
                                `[CRM Phone AutoSync] Branch ${branchId} unavailable: ${error?.message || error}`
                            );

                            return null;
                        }
                    }
                )
            );

        const successful =
            results
                .filter(
                    result =>
                        result.status ===
                        "fulfilled" &&
                        result.value
                )
                .map(
                    result =>
                        result.value
                );

        if (
            successful.length > 0
        ) {

            console.log(
                "[CRM Phone AutoSync] Completed:",
                successful
            );
        }

    } catch (
        error
    ) {

        console.warn(
            "[CRM Phone AutoSync] General error:",
            error?.message || error
        );

    } finally {

        crmPhoneSyncRunning =
            false;
    }
}


function startCrmPhoneAutoSync() {

    if (
        crmPhoneSyncTimer
    ) {
        return;
    }

    console.log(
        "[CRM Phone AutoSync] Started. Interval = 120000 ms."
    );

    setTimeout(
        () => {
            void runCrmPhoneAutoSync();
        },
        30000
    );

    crmPhoneSyncTimer =
        setInterval(
            () => {
                void runCrmPhoneAutoSync();
            },
            120000
        );
}

import Database from "better-sqlite3";
import path from "node:path";
import {
    app
} from "electron";


let database =
    null;


/*
 * =========================================================
 * DATABASE PATH
 * =========================================================
 */

function getDatabasePath() {

    return path.join(

        app.getPath(
            "userData"
        ),

        "pharmabi-local.db"

    );

}


/*
 * =========================================================
 * OPEN DATABASE
 * =========================================================
 */

function openDatabase() {

    if (
        database
    ) {

        return database;

    }


    database =
        new Database(
            getDatabasePath()
        );


    database.pragma(
        "journal_mode = WAL"
    );


    database.pragma(
        "synchronous = NORMAL"
    );


    database.pragma(
        "foreign_keys = ON"
    );


    database.pragma(
        "busy_timeout = 10000"
    );


    createSchema();


    console.log(

        `[localSalesDb] OPEN ${getDatabasePath()}`

    );


    return database;

}

 * =========================================================
 * SCHEMA

 */

            last_sync_rows INTEGER DEFAULT 0,

            total_rows INTEGER DEFAULT 0,

            error_message TEXT

        );

    `);

    const salesHistoryColumns =
        db.prepare(
            "PRAGMA table_info(sales_history)"
        ).all();

    if (
        !salesHistoryColumns.some(
            column => column.name === "pin_number"
        )
    ) {

        db.exec(
            "ALTER TABLE sales_history ADD COLUMN pin_number TEXT"
        );

    }

}

function getExclusiveDate(

    value

) {

    const normalized =
        String(
            value ||
            ""
        ).trim();


    if (
        !/^\d{4}-\d{2}-\d{2}$/.test(
            normalized
        )
    ) {

        throw new Error(
            "Invalid date. Expected YYYY-MM-DD"
        );

    }


    const [
        year,
        month,
        day
    ] =
        normalized
            .split(
                "-"
            )
            .map(
                Number
            );


    const next =
        new Date(

            Date.UTC(

                year,
                month - 1,
                day + 1

            )

        );


    return [

        next.getUTCFullYear(),

        String(
            next.getUTCMonth() + 1
        ).padStart(
            2,
            "0"
        ),

        String(
            next.getUTCDate()
        ).padStart(
            2,
            "0"
        )

    ].join(
        "-"
    );

}

/*
 * =========================================================
 * SAVE ROWS
 * =========================================================
 */

export function saveSalesRows(
    {
        branchId,
        rows,
        batchSize = 5000
    }
) {

    const id =
        Number(
            branchId
        );


    if (
        !Number.isInteger(
            id
        ) ||
        id <= 0
    ) {

        throw new Error(
            "Invalid branchId"
        );

    }


    if (
        !Array.isArray(
            rows
        ) ||
        rows.length === 0
    ) {

        return {

            branchId:
                id,

            processed:
                0

        };

    }


    const db =
        openDatabase();


    const statements =
        getStatements();


    const normalizedBatchSize =
        Number.isInteger(
            Number(
                batchSize
            )
        ) &&
        Number(
            batchSize
        ) > 0

            ? Math.min(
                Number(
                    batchSize
                ),
                10000
            )

            : 5000;


    let processed =
        0;


    const insertBatch =
        db.transaction(

            batch => {

                for (
                    const row
                    of batch
                ) {

                    if (
                        row?.lineId ===
                        null ||
                        row?.lineId ===
                        undefined ||
                        String(
                            row.lineId
                        ) ===
                        ""
                    ) {

                        continue;

                    }


                    statements
                        .insertOrReplace
                        .run(

                            normalizeRow(
                                id,
                                row
                            )

                        );


                    processed +=
                        1;

                }

            }

        );


    for (
        let start = 0;

        start < rows.length;

        start += normalizedBatchSize

    ) {

        insertBatch(

            rows.slice(

                start,

                start +
                normalizedBatchSize

            )

        );

    }


    return {

        branchId:
            id,

        processed

    };

}


/*
 * =========================================================
 * GET SALES RANGE
 * =========================================================
 */

export function getSalesRows(
    {
        branchId,
        fromDate,
        toDate
    }
) {

    const id =
        Number(
            branchId
        );


    if (
        !Number.isInteger(
            id
        ) ||
        id <= 0
    ) {

        throw new Error(
            "Invalid branchId"
        );

    }


    const from =
        String(
            fromDate ||
            ""
        ).trim();


    const to =
        String(
            toDate ||
            ""
        ).trim();


    if (
        !/^\d{4}-\d{2}-\d{2}$/.test(
            from
        )
    ) {

        throw new Error(
            "Invalid fromDate. Expected YYYY-MM-DD"
        );

    }


    if (
        !/^\d{4}-\d{2}-\d{2}$/.test(
            to
        )
    ) {

        throw new Error(
            "Invalid toDate. Expected YYYY-MM-DD"
        );

    }


    const toExclusive =
        getExclusiveDate(
            to
        );


    const startedAt =
        Date.now();


    const rows =
        getStatements()
            .getSalesRange
            .all(

                id,

                `${from}T00:00:00.000Z`,

                `${toExclusive}T00:00:00.000Z`

            );


    const elapsed =
        Date.now() -
        startedAt;


    console.log(

        `[localSalesDb] READ branch=${id} range=${from}|${to} rows=${rows.length} in ${elapsed} ms`

    );


    return rows;

}

 */

export function hasSalesMonth(
    {
        branchId,
        month
    }
) {

    const id =
        Number(
            branchId
        );


    const normalizedMonth =
        String(
            month ||
            ""
        );


    if (
        !/^\d{4}-\d{2}$/.test(
            normalizedMonth
        )
    ) {

        return false;

    }


    const fromDate =
        `${normalizedMonth}-01`;


    const [
        year,
        monthNumber
    ] =
        normalizedMonth
            .split(
                "-"
            )
            .map(
                Number
            );


    const nextMonth =
        new Date(

            Date.UTC(

                year,

                monthNumber,

                1

            )

        );


    const toDate = [

        nextMonth.getUTCFullYear(),

        String(
            nextMonth.getUTCMonth() + 1
        ).padStart(
            2,
            "0"
        ),

        "01"

    ].join(
        "-"
    );


    const result =
        getStatements()
            .getMonthStats
            .get(

                id,

                fromDate,

                toDate

            );


    return (
        Number(
            result?.total_rows
        ) || 0
    ) > 0;

}

 * SYNC STATE
 * =========================================================
 */

export function updateSalesSyncState(
    {
        branchId,
        status,
        rows = 0,
        errorMessage = null
    }
) {

    const id =
        Number(
            branchId
        );


    const statements =
        getStatements();


    const range =
        statements
            .getDateRange
            .get(
                id
            );


    const existing =
        statements
            .getSyncState
            .get(
                id
            );


    const now =
        new Date()
            .toISOString();


    const startedAt =
        status === "RUNNING"

            ? now

            : (
                existing?.last_sync_started_at ||
                null
            );


    const finishedAt =
        status === "READY" ||
        status === "ERROR"

            ? now

            : (
                existing?.last_sync_finished_at ||
                null
            );


    statements
        .saveSyncState
        .run({

            branch_id:
                id,

            oldest_date:
                range?.oldest_date ||
                null,

            newest_date:
                range?.newest_date ||
                null,

            last_sync_started_at:
                startedAt,

            last_sync_finished_at:
                finishedAt,

            last_sync_status:
                String(
                    status ||
                    "UNKNOWN"
                ),

            last_sync_rows:
                Number(
                    rows
                ) || 0,

            total_rows:
                Number(
                    range?.total_rows
                ) || 0,

            error_message:
                errorMessage
                ? String(
                    errorMessage
                )
                : null

        });


    return statements
        .getSyncState
        .get(
            id
        );

}
